    async function checkCacheReadiness() {

      if (!('caches' in window)) return false;

      try {

        const cache = await caches.open(CACHE_NAME);

        const keys = await cache.keys();

        return keys.length >= 3;

      } catch (e) {

        return false;

      }

    }


    async function precacheAssets() {

      if ('caches' in window) {

        try {

          const cache = await caches.open(CACHE_NAME);

          await cache.addAll(CORE_ASSETS);

          updateOnlineStatus();

        } catch (err) {

          console.warn('Pre-caching background note:', err);

        }

      }

    }


    async function updateOnlineStatus() {

      const banner = document.getElementById('offlineBanner');

      const dot = document.getElementById('offlineIndicatorDot');

      const isOnline = navigator.onLine;


      if (isOnline) {

        banner.classList.add('hidden');

        if (dot) {

          dot.style.color = '#10B981';

          dot.title = "Online (Live): Fully connected";

        }

      } else {

        banner.classList.remove('hidden');

        const hasCache = await checkCacheReadiness();

        if (dot) {

          if (hasCache) {

            dot.style.color = '#06B6D4';

            dot.title = "Offline (Cached & Ready): Running locally";

          } else {

            dot.style.color = '#64748B';

            dot.title = "Offline (Not Cached): Disconnected";

          }

        }

      }

    }
